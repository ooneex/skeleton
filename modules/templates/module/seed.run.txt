#!/usr/bin/env bun

import { join } from "node:path";
import { loadEnv } from "@talosjs/app-env";
import { run } from "@talosjs/seeds";
// Every module, so the whole entity graph is registered on the main data
// source — a seed reads relations that cross module boundaries, and TypeORM
// only maps one once both sides are registered.
import "@module/app/AppModule";
import "@module/{{name}}/seeds/seeds";

await loadEnv([join(import.meta.dir, "../../../app/.env.yml")]);

await run();
