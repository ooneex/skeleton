import {
  DataSource,
  decorator,
  type CloudflareDataSourceOptionsType,
  type ICloudflareDatabase,
} from "@talosjs/database";
import { SharedModule } from "@module/shared/SharedModule";

@decorator.database()
export class {{NAME}}Database {
  private readonly source: DataSource<CloudflareDataSourceOptionsType>;

  public constructor(client: ICloudflareDatabase) {
    this.source = new DataSource({
      type: "cloudflare",
      client,
      entities: SharedModule.entities,
      synchronize: false,
    });
  }

  public getSource(): DataSource<CloudflareDataSourceOptionsType> {
    return this.source;
  }

  public getClient(): ICloudflareDatabase {
    return this.source.client;
  }

  public async open(): Promise<void> {
    if (!this.source.isInitialized) {
      await this.source.initialize();
    }
  }

  public async close(): Promise<void> {
    if (this.source.isInitialized) {
      await this.source.destroy();
    }
  }

  public async drop(): Promise<void> {
    if (this.source.isInitialized) {
      await this.source.dropDatabase();
    }
  }
}
