import { DataSource, DatabaseException, SqlDatabase, decorator } from "@talosjs/database";
import { inject } from "@talosjs/container";
import { AppEnv } from "@talosjs/app-env";
import { SharedModule } from "@module/shared/SharedModule";

@decorator.database()
export class {{NAME}}Database extends SqlDatabase {
  public constructor(@inject(AppEnv) private readonly env: AppEnv) {
    super();
  }

  public getSource(database?: string): DataSource {
    if (this.source) {
      return this.source;
    }

    const url = this.env.DATABASE_URL;

    if (!url) {
      throw new DatabaseException(
        "ClickHouse URL is required. Set the DATABASE_URL environment variable.",
        "CONNECTION_FAILED",
      );
    }

    this.source = new DataSource({
      synchronize: false,
      entities: SharedModule.entities,
      poolSize: 10,
      database,
      url,
      type: "clickhouse",
    });

    return this.source;
  }
}
